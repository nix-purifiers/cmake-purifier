cmake_minimum_required(VERSION 2.8)
project(leaf C)
execute_process(COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/identity.py" "${CMAKE_COMMAND}" "${CMAKE_ROOT}" "${CMAKE_VERSION}" "${CMAKE_BINARY_DIR}/leaf-identity.json" RESULT_VARIABLE identity_status)
if(NOT identity_status EQUAL 0)
  message(FATAL_ERROR "native leaf identity capture failed")
endif()
add_library(leaf STATIC leaf.c)
install(TARGETS leaf ARCHIVE DESTINATION lib)
install(FILES "${CMAKE_BINARY_DIR}/leaf-identity.json" DESTINATION share/legacy-proof)
